# Function to configure common target properties for HIP targets
function(flashmoe_hip_target_common tgt)
    # Link against HIP runtime and AMD libraries
    target_link_libraries(${tgt} PRIVATE
        hip::device
        hip::host
        roc::hipcub
        roc::rocprim
        roc::rocwmma
        roc::hipblaslt
    )

    # Link atomic library
    target_link_libraries(${tgt} PRIVATE atomic)

    # rocSHMEM if available
    if(EXISTS "${ROCSHMEM_ROOT}/lib/librocshmem.so")
        target_link_libraries(${tgt} PRIVATE rocshmem)
    endif()

    # Set HIP-specific compile options
    target_compile_options(${tgt} PRIVATE
        $<$<COMPILE_LANGUAGE:HIP>:-fgpu-rdc>
        $<$<COMPILE_LANGUAGE:HIP>:-Wno-unused-result>
    )

    # Add architecture-specific options
    foreach(GPU_TARGET ${GPU_TARGETS})
        target_compile_options(${tgt} PRIVATE
            $<$<COMPILE_LANGUAGE:HIP>:--offload-arch=${GPU_TARGET}>
        )
    endforeach()

    # Debug vs Release flags
    target_compile_options(${tgt} PRIVATE
        $<$<AND:$<CONFIG:Debug>,$<COMPILE_LANGUAGE:CXX>>:-O0;-g>
        $<$<AND:$<CONFIG:Debug>,$<COMPILE_LANGUAGE:HIP>>:-O0;-g;-ggdb>
        $<$<AND:$<CONFIG:Release>,$<COMPILE_LANGUAGE:CXX>>:-O3>
        $<$<AND:$<CONFIG:Release>,$<COMPILE_LANGUAGE:HIP>>:-O3>
    )

    # Target include directories
    target_include_directories(${tgt} PRIVATE
        ${CMAKE_CURRENT_SOURCE_DIR}/include
        ${ROCM_SDK_ROOT}/include
        ${ROCSHMEM_ROOT}/include
    )

    # Compile definitions
    target_compile_definitions(${tgt} PRIVATE
        __HIP_PLATFORM_AMD__
        FLASHMOE_HIP=1
    )
endfunction()

# Function for test targets with minimal dependencies

        hip::device
        hip::host
        roc::hipcub
        roc::rocprim
    )

    # Set HIP-specific compile options
    target_compile_options(${tgt} PRIVATE
        $<$<COMPILE_LANGUAGE:HIP>:-fgpu-rdc>
    )

    # Add architecture-specific options
    foreach(GPU_TARGET ${GPU_TARGETS})
        target_compile_options(${tgt} PRIVATE
            $<$<COMPILE_LANGUAGE:HIP>:--offload-arch=${GPU_TARGET}>
        )
    endforeach()

    # Debug vs Release flags
    target_compile_options(${tgt} PRIVATE
        $<$<AND:$<CONFIG:Debug>,$<COMPILE_LANGUAGE:CXX>>:-O0;-g>
        $<$<AND:$<CONFIG:Debug>,$<COMPILE_LANGUAGE:HIP>>:-O0;-g;-ggdb>
        $<$<AND:$<CONFIG:Release>,$<COMPILE_LANGUAGE:CXX>>:-O3>
        $<$<AND:$<CONFIG:Release>,$<COMPILE_LANGUAGE:HIP>>:-O3>
    )

    target_compile_definitions(${tgt} PRIVATE
        __HIP_PLATFORM_AMD__
        FLASHMOE_HIP=1
    )
endfunction()
